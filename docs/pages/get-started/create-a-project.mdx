from binance import AsyncClient, BinanceSocketManager
import asyncio
import pandas as pd

symbol = "BTCUSDT"
h4_data = pd.DataFrame()
h1_data = pd.DataFrame()

async def main():
    client = await AsyncClient.create()
    bm = BinanceSocketManager(client)

    # 1h kline socket
    h1_socket = bm.kline_socket(symbol=symbol, interval="1h")
    # 4h kline socket
    h4_socket = bm.kline_socket(symbol=symbol, interval="4h")

    async with h1_socket as h1_stream, h4_socket as h4_stream:
        while True:
            res1 = await h1_stream.recv()
            res4 = await h4_stream.recv()

            handle_kline(res1, "H1")
            handle_kline(res4, "H4")

def handle_kline(msg, timeframe):
    k = msg['k']
    if k['x']:  # candle closed
        candle = {
            "open": float(k['o']),
            "high": float(k['h']),
            "low": float(k['l']),
            "close": float(k['c'])
        }
        print(f"{timeframe} closed candle:", candle)

asyncio.run(main())
def detect_market_structure(df):
    highs = df['high']
    lows = df['low']

    if highs.iloc[-1] > highs.iloc[-2] and lows.iloc[-1] > lows.iloc[-2]:
        return "Bullish"
    elif highs.iloc[-1] < highs.iloc[-2] and lows.iloc[-1] < lows.iloc[-2]:
        return "Bearish"
    else:
        return "Neutral"
def find_swings(df):
    swings = []
    for i in range(2, len(df)-2):
        if df['low'][i] < df['low'][i-1] and df['low'][i] < df['low'][i+1]:
            swings.append(("low", i))
        if df['high'][i] > df['high'][i-1] and df['high'][i] > df['high'][i+1]:
            swings.append(("high", i))
    return swings
def detect_double_bottom(df):
    swings = find_swings(df)
    lows = [i for t, i in swings if t == "low"]

    if len(lows) >= 2:
        last_two = lows[-2:]
        price1 = df['low'][last_two[0]]
        price2 = df['low'][last_two[1]]

        if abs(price1 - price2) / price1 < 0.01:  # 1% tolerance
            return True
    return False
def breakout_retest(df):
    resistance = df['high'][-20:].max()
    support = df['low'][-20:].min()
    close = df['close'].iloc[-1]

    if close > resistance:
        return "Bullish Breakout"

    if close < support:
        return "Bearish Breakout"

    return None
